la creation des tables commandes,articles et lignes de commande:
 CREATE TABLE commandes(

  Date_commande DATE
);
 CREATE TABLE articles(
 NUM_article int PRIMARY KEY,
 Nom_article VARCHAR(12) UNIQUE, 
Prix_vente NUMBER(6,2) 
 );
CREATE TABLE lignes_commande ( 
Num_article INT,  
Num_commande CHAR(8), 
qantite_commande INT NOT NULL,
CONSTRAINT FK_Num_art FOREIGN KEY(Num_article) REFERENCES articles( NUM_article),
CONSTRAINT FK_Num_CMD FOREIGN KEY( Num_commande) REFERENCES commandes( Num_commande),
CONSTRAINT PK_LC PRIMARY KEY (Num_article,Num_commande) 
 );
